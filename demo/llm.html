# briefing.py — same bullets as the page. No network.
def brief_workforce(snapshot: dict, department: str | None = None) -> list[str]:
    rows = snapshot["departments"]
    if department:
        rows = [r for r in rows if r["name"] == department]
    hc = sum(r["headcount"] for r in rows)
    open_roles = sum(r["open_roles"] for r in rows)
    top = max(rows, key=lambda r: r["overtime_hours_12mo"])
    # five bullets: cut, vacancy, trend, overtime, next check
    return five_bullets(snapshot, rows, hc, open_roles, top)
